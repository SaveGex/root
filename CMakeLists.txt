cmake_minimum_required(VERSION 3.16)
project(EquationSolver VERSION 0.1.0 LANGUAGES CXX)

find_package(Boost CONFIG REQUIRED COMPONENTS regex)
find_package(muparser CONFIG REQUIRED)

add_library(equation_solver root.cpp Linear_Equation.cpp Quadratic_Equations.cpp Root_from_Root.cpp)
target_compile_features(equation_solver PUBLIC cxx_std_17)
set_target_properties(equation_solver PROPERTIES CXX_EXTENSIONS OFF)
target_include_directories(equation_solver PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(equation_solver PRIVATE Boost::regex muparser::muparser)
if(MSVC)
    target_compile_options(equation_solver PRIVATE /W4 /permissive- /utf-8)
else()
    target_compile_options(equation_solver PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(equation-solver main.cpp)
target_link_libraries(equation-solver PRIVATE equation_solver)

include(CTest)
if(BUILD_TESTING)
    add_executable(solver-tests tests/solver_tests.cpp)
    target_link_libraries(solver-tests PRIVATE equation_solver)
    add_test(NAME solver-regressions COMMAND solver-tests)
    add_test(NAME cli-demo COMMAND equation-solver)
    add_test(NAME cli-quadratic COMMAND equation-solver quadratic "x^2 - 3*x + 2 = 0")
    set_tests_properties(cli-quadratic PROPERTIES PASS_REGULAR_EXPRESSION "x1 = 1, x2 = 2")
    add_test(NAME cli-invalid-input COMMAND equation-solver quadratic "not an equation")
    set_tests_properties(cli-invalid-input PROPERTIES WILL_FAIL TRUE)
endif()
